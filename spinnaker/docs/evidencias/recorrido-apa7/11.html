Paso 11. Service: exposición mediante NodePort

Archivo versionado · Lectura documental: 2026-11-20T05:12:54.301376+00:00 UTC

service.yml

apiVersion: v1
kind: Service
metadata:
  name: sarseg-web
  namespace: sarseg-dev
  labels:
    app: sarseg-web
spec:
  type: NodePort
  selector:
    app: sarseg-web
  ports:
    - name: http
      protocol: TCP
      port: 80
      targetPort: http
      nodePort: 30380
